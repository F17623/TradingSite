| CREATE OR REPLACE FUNCTION public.recalc_risk()<br> RETURNS void<br> LANGUAGE plpgsql<br> SECURITY DEFINER<br> SET search_path TO 'public'<br>AS $function$<br>declare<br>  v_total numeric := 0; v_peak numeric := 0; v_count int := 0; v_last bigint;<br>  v_loss numeric; v_profit numeric; v_pb boolean;<br>  v_unlock timestamptz; v_kind text; v_af timestamptz; v_au timestamptz;<br>  v_tstart numeric; v_tgive numeric; v_maxtr int; v_tday date;<br>  v_today date := ((now() at time zone 'America/Chicago') + interval '7 hours')::date;<br>  v_stop boolean := false; v_type text := ''; v_text text := '';<br>  v_trail_now boolean; v_trail_active boolean;<br>  -- скользящий лимит<br>  v_win int; v_brk int;<br>  v_all timestamptz[]; v_n int := 0; v_c int := 0;<br>  v_rate_used int := 0;<br>  v_free_at timestamptz; v_rate_until timestamptz; v_brk_until timestamptz;<br>begin<br>  insert into public.risk_control (id, full_stop, limit_type) values (1, false, '') on conflict (id) do nothing;<br><br>  select loss_limit, profit_limit,<br>         unlock_at, manual_kind, asia_from, asia_until,<br>         trail_start_pct, trail_giveback_pct, max_trades, trail_day,<br>         rate_window_min, rate_break_min<br>    into v_loss, v_profit, v_unlock, v_kind, v_af, v_au,<br>         v_tstart, v_tgive, v_maxtr, v_tday,<br>         v_win, v_brk<br>  from public.risk_control where id = 1;<br>  v_pb := false;   -- режима Playback больше нет: считаются только настоящие счета<br><br>  -- реализованный итог дня и его пик (по закрытым позициям, в порядке записи)<br>  with flat as (<br>    select id, account_name, instrument,<br>           sum(-side * qty * price * point_value -
