(function() {
    const originalFetch = window.fetch;
    window.fetch = async function(...args) {
        if (typeof args[0] === 'string' && args[0].includes('trade_shots')) {
            console.log('🔍 ПЕРЕХВАЧЕН ЗАПРОС К trade_shots:', args[0]);
            if (args[1] && args[1].body) {
                console.log('📦 ТЕЛО ЗАПРОСА (Payload):', JSON.parse(args[1].body));
            }
        }
        const response = await originalFetch.apply(this, args);
        if (typeof args[0] === 'string' && args[0].includes('trade_shots')) {
            console.log('📥 СТАТУС ОТВЕТА:', response.status, response.statusText);
            const clone = response.clone();
            clone.text().then(txt => {
                if (txt) console.log('💬 ТЕКСТ ОТВЕТА СЕРВЕРА:', txt);
            });
        }
        return response;
    };
    console.log("✅ Перехватчик запросов trade_shots успешно активирован!");
})();
