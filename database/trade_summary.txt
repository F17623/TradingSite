create view public.trade_summary
with
  (security_invoker = on) as
select
  trade_no,
  group_no,
  account_name,
  instrument,
  case
    when (
      array_agg(
        side
        order by
          id
      )
    ) [1] > 0 then 'LONG'::text
    else 'SHORT'::text
  end as direction,
  min(created_at) as opened_at,
  case
    when (
      array_agg(
        net_after
        order by
          id desc
      )
    ) [1] = 0 then max(created_at)
    else null::timestamp with time zone
  end as closed_at,
  (
    array_agg(
      net_after
      order by
        id desc
    )
  ) [1] as net_now,
  max(abs(net_after)) as max_qty,
  sum(
    ((- side) * qty)::numeric * price * point_value - COALESCE(commission, 0::numeric)
  ) as cash
from
  trades
where
  trade_no is not null
group by
  trade_no,
  group_no,
  account_name,
  instrument;
